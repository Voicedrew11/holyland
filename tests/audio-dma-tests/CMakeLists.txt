cmake_minimum_required(VERSION 3.20)
project(kfiv_audio_dma_interrupt_tests LANGUAGES CXX)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/PackagePaths.cmake")

set(PS2X_IOP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
add_subdirectory("${PS2RECOMP_DIR}/ps2xIOP" iop)
add_executable(kfiv_audio_dma_interrupt_tests audio_dma_tests.cpp)
target_link_libraries(kfiv_audio_dma_interrupt_tests PRIVATE ps2x::iop)
target_compile_features(kfiv_audio_dma_interrupt_tests PRIVATE cxx_std_20)

# A controlled regression build uses the very same emulator and hardware
# sources, removing only the cancelled-channel dispatch guard. No production
# source is modified. The fixture must expose the original guest IRQ/DMA loop.
get_target_property(IOP_SOURCES ps2_iop SOURCES)
set(CONTROL_SOURCES)
foreach(IOP_SOURCE IN LISTS IOP_SOURCES)
    if(IOP_SOURCE STREQUAL "src/emulator/iop_emulator.cpp")
        file(READ "${PS2RECOMP_DIR}/ps2xIOP/${IOP_SOURCE}" CONTROL_EMULATOR)
        string(REPLACE "\r\n" "\n" CONTROL_EMULATOR "${CONTROL_EMULATOR}")
        set(CANCEL_GUARD "                    if (!memory.soundDmaActive(irq))\n                        continue;")
        string(FIND "${CONTROL_EMULATOR}" "${CANCEL_GUARD}" GUARD_POSITION)
        if(GUARD_POSITION LESS 0)
            message(FATAL_ERROR "Cancelled sound-DMA guard changed; review the regression control")
        endif()
        string(REPLACE "${CANCEL_GUARD}" "                    // Regression control: old cancelled IRQ is delivered." CONTROL_EMULATOR "${CONTROL_EMULATOR}")
        file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/control_iop_emulator.cpp" "${CONTROL_EMULATOR}")
        list(APPEND CONTROL_SOURCES "${CMAKE_CURRENT_BINARY_DIR}/control_iop_emulator.cpp")
    else()
        list(APPEND CONTROL_SOURCES "${PS2RECOMP_DIR}/ps2xIOP/${IOP_SOURCE}")
    endif()
endforeach()
add_library(ps2_iop_dma_cancel_control STATIC ${CONTROL_SOURCES})
target_compile_features(ps2_iop_dma_cancel_control PUBLIC cxx_std_20)
target_include_directories(ps2_iop_dma_cancel_control PUBLIC "${PS2RECOMP_DIR}/ps2xIOP/include"
    PRIVATE "${PS2RECOMP_DIR}/ps2xIOP/src" "${PS2RECOMP_DIR}/ps2xIOP/src/emulator")
add_executable(kfiv_audio_dma_interrupt_control audio_dma_tests.cpp)
target_link_libraries(kfiv_audio_dma_interrupt_control PRIVATE ps2_iop_dma_cancel_control)
target_compile_features(kfiv_audio_dma_interrupt_control PRIVATE cxx_std_20)
if(MSVC)
    target_compile_options(kfiv_audio_dma_interrupt_tests PRIVATE /W4 /permissive-)
    target_compile_options(kfiv_audio_dma_interrupt_control PRIVATE /W4 /permissive-)
    target_compile_definitions(ps2_iop_dma_cancel_control PRIVATE _CRT_SECURE_NO_WARNINGS)
endif()
enable_testing()
add_test(NAME actual_guest_dma_stop COMMAND kfiv_audio_dma_interrupt_tests)
add_test(NAME original_stale_irq_control COMMAND kfiv_audio_dma_interrupt_control --expect-leak)
