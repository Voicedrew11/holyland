cmake_minimum_required(VERSION 3.21)
project(kfiv_process_audio_capture LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Process audio loopback requires Windows 10 build 20348 or newer")
endif()
add_executable(kfiv_process_audio_capture process_audio_capture.cpp)
target_compile_features(kfiv_process_audio_capture PRIVATE cxx_std_20)
target_compile_definitions(kfiv_process_audio_capture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_link_libraries(kfiv_process_audio_capture PRIVATE ole32 mmdevapi)
if(MSVC)
    target_compile_options(kfiv_process_audio_capture PRIVATE /W4 /permissive-)
endif()
