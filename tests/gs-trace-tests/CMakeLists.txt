cmake_minimum_required(VERSION 3.21)
project(gs_trace_tests LANGUAGES CXX)
set(PS2RECOMP_DIR "" CACHE PATH "Runtime checkout to test")
set(PRIOR_TRACE_SOURCE "" CACHE FILEPATH "Optional pre-fix trace source for negative controls")
set(GS_DIR "${PS2RECOMP_DIR}/ps2xRuntime/src/lib/gs")
if(NOT EXISTS "${GS_DIR}/gs_trace.cpp")
    message(FATAL_ERROR "Set PS2RECOMP_DIR to the runtime checkout")
endif()
function(add_trace_test target source)
    add_executable(${target} trace_tests.cpp "${source}")
    target_include_directories(${target} PRIVATE "${PS2RECOMP_DIR}/ps2xRuntime/include" "${GS_DIR}")
    target_compile_features(${target} PRIVATE cxx_std_20)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /fp:strict)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
endfunction()
add_trace_test(gs_trace_tests "${GS_DIR}/gs_trace.cpp")
enable_testing()
foreach(CASE range register_only transfer_only draw_only crt2 fraction vram_after_to vram_only disabled)
    add_test(NAME trace.${CASE} COMMAND gs_trace_tests ${CASE} "${CMAKE_CURRENT_BINARY_DIR}/${CASE}")
    set_tests_properties(trace.${CASE} PROPERTIES TIMEOUT 10)
endforeach()
if(PRIOR_TRACE_SOURCE)
    if(NOT EXISTS "${PRIOR_TRACE_SOURCE}")
        message(FATAL_ERROR "PRIOR_TRACE_SOURCE does not exist")
    endif()
    add_trace_test(gs_trace_prior_control "${PRIOR_TRACE_SOURCE}")
    foreach(CASE range register_only transfer_only draw_only crt2 fraction)
        add_test(NAME trace.prior_${CASE} COMMAND gs_trace_prior_control ${CASE} "${CMAKE_CURRENT_BINARY_DIR}/prior_${CASE}")
        set_tests_properties(trace.prior_${CASE} PROPERTIES TIMEOUT 10 WILL_FAIL TRUE)
    endforeach()
endif()
