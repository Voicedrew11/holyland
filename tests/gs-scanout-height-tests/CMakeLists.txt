cmake_minimum_required(VERSION 3.21)
project(kfiv_gs_scanout_height_tests LANGUAGES CXX)

set(PS2RECOMP_DIR "" CACHE PATH "Patched PS2Recomp checkout")
if(NOT PS2RECOMP_DIR OR NOT EXISTS "${PS2RECOMP_DIR}/ps2xRuntime/src/lib/gs/gs_cpu_backend.cpp")
    message(FATAL_ERROR "Set PS2RECOMP_DIR to a patched PS2Recomp checkout")
endif()
set(GS_SOURCE_DIR "${PS2RECOMP_DIR}/ps2xRuntime/src/lib/gs")
set(GS_SCANOUT_PRIOR_SOURCE "" CACHE FILEPATH "Optional pre-height-fix backend for failing controls")
find_package(Threads REQUIRED)

function(add_scanout_target target source)
    add_executable(${target} gs_scanout_height_tests.cpp "${source}" "${GS_SOURCE_DIR}/ps2_gs_memory.cpp")
    if(EXISTS "${GS_SOURCE_DIR}/gs_trace.cpp")
        target_sources(${target} PRIVATE "${GS_SOURCE_DIR}/gs_trace.cpp")
    endif()
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_include_directories(${target} PRIVATE "${PS2RECOMP_DIR}/ps2xRuntime/include" "${GS_SOURCE_DIR}")
    target_link_libraries(${target} PRIVATE Threads::Threads)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /permissive- /bigobj /fp:strict)
        target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS)
    endif()
endfunction()

add_scanout_target(kfiv_gs_scanout_height_tests "${GS_SOURCE_DIR}/gs_cpu_backend.cpp")
enable_testing()
set(scanout_cases crt2_game crt1_game dual_crt magnification odd_rounding frame_clamp progressive field_rows small_frame)
foreach(case IN LISTS scanout_cases)
    add_test(NAME kfiv_gs_scanout_${case} COMMAND kfiv_gs_scanout_height_tests ${case})
    set_tests_properties(kfiv_gs_scanout_${case} PROPERTIES
        ENVIRONMENT "PS2X_GS_THREADS=1;PS2X_NO_WEAVE=0" TIMEOUT 60)
endforeach()
if(GS_SCANOUT_PRIOR_SOURCE)
    if(NOT EXISTS "${GS_SCANOUT_PRIOR_SOURCE}")
        message(FATAL_ERROR "GS_SCANOUT_PRIOR_SOURCE does not exist")
    endif()
    add_scanout_target(kfiv_gs_scanout_height_prior_control "${GS_SCANOUT_PRIOR_SOURCE}")
    foreach(case crt2_game crt1_game dual_crt magnification odd_rounding frame_clamp)
        add_test(NAME kfiv_gs_scanout_prior_${case} COMMAND kfiv_gs_scanout_height_prior_control ${case})
        set_tests_properties(kfiv_gs_scanout_prior_${case} PROPERTIES
            ENVIRONMENT "PS2X_GS_THREADS=1;PS2X_NO_WEAVE=0" WILL_FAIL TRUE TIMEOUT 60)
    endforeach()
endif()
