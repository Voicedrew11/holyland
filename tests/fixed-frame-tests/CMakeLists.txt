cmake_minimum_required(VERSION 3.21)
project(fixed_frame_tests LANGUAGES CXX)
set(PS2RECOMP_DIR "" CACHE PATH "Patched runtime source")
if(NOT EXISTS "${PS2RECOMP_DIR}/ps2xRuntime/include/runtime/fixed_frame_pacer.h")
    message(FATAL_ERROR "Set PS2RECOMP_DIR to the patched source checkout")
endif()
add_executable(fixed_frame_pacer pacer_tests.cpp)
target_compile_features(fixed_frame_pacer PRIVATE cxx_std_20)
target_include_directories(fixed_frame_pacer PRIVATE "${PS2RECOMP_DIR}/ps2xRuntime/include")
enable_testing()
add_test(NAME fixed_frame_pacer COMMAND fixed_frame_pacer)

set(RUNTIME_BUILD_DIR "" CACHE PATH "Existing native runtime build for integration tests")
if(RUNTIME_BUILD_DIR)
    include(runtime_link.cmake)
    add_fixed_runtime_test(fixed_frame_runtime)
    add_test(NAME fixed_frame_runtime COMMAND fixed_frame_runtime)
    add_fixed_runtime_test(fixed_frame_interpolation interpolation_tests.cpp)
    add_test(NAME fixed_frame_interpolation COMMAND fixed_frame_interpolation)
    set_tests_properties(fixed_frame_runtime PROPERTIES TIMEOUT 10)
endif()
