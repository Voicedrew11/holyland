cmake_minimum_required(VERSION 3.21)
project(cooperative_runtime_tests LANGUAGES C CXX)
set(PS2RECOMP_DIR "" CACHE PATH "Full patched PS2Recomp checkout")
set(COOPERATIVE_GS_BACKEND "cpu" CACHE STRING "Explicit CPU or Vulkan functional backend")
set_property(CACHE COOPERATIVE_GS_BACKEND PROPERTY STRINGS cpu vulkan)
if(NOT COOPERATIVE_GS_BACKEND MATCHES "^(cpu|vulkan)$")
    message(FATAL_ERROR "Use cpu or vulkan explicitly; auto fallback is not proof")
endif()
option(COOPERATIVE_ENABLE_OBSOLETE_Q_CONTROL "Run the retained, obsolete Q physical-clock assertion" OFF)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/CooperativeRuntimeFixtures.cmake")
cooperative_require_runtime()
set(SOURCE "${PS2RECOMP_DIR}")
set(PHYSICAL "${CMAKE_CURRENT_SOURCE_DIR}/physical")
set(COMPAT "${CMAKE_CURRENT_SOURCE_DIR}/compatibility")
function(link_fixture target fixture)
    cooperative_runtime_fixture(${target} "${fixture}")
endfunction()
function(pure_fixture target fixture)
    add_executable(${target} "${fixture}")
    cooperative_fixture_flags(${target})
    target_include_directories(${target} PRIVATE "${SOURCE}/ps2xRuntime/include")
endfunction()
file(GLOB_RECURSE fixture_source_inputs CONFIGURE_DEPENDS LIST_DIRECTORIES false
    "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp" "${CMAKE_CURRENT_SOURCE_DIR}/*.py")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${fixture_source_inputs}
    "${SOURCE}/ps2xRuntime/include/runtime/ee_scheduler.h"
    "${SOURCE}/ps2xRuntime/include/runtime/vblank_field_pacer.h"
    "${SOURCE}/ps2xRuntime/src/lib/Kernel/EeScheduler.cpp")
enable_testing()
pure_fixture(field_pacing "${COMPAT}/field-pacing-tests/field_pacing_tests.cpp")
add_test(NAME cooperative_clock_helper_legacy_methods COMMAND field_pacing)
add_test(NAME cooperative_clock_helper_rebase_control COMMAND field_pacing --legacy-control)
add_test(NAME cooperative_clock_helper_grid_control COMMAND field_pacing --grid-control)
pure_fixture(ntsc_pacing "${COMPAT}/field-pacing-tests/ntsc_pacer_tests.cpp")
add_test(NAME cooperative_clock_ntsc_legacy_bounds COMMAND ntsc_pacing)
pure_fixture(physical_pure "${PHYSICAL}/pure_tests.cpp")
add_test(NAME cooperative_clock_physical_pure COMMAND physical_pure)

link_fixture(vblank "${COMPAT}/vblank-tests/vblank_tests.cpp")
add_test(NAME cooperative_clock_vblank_basic COMMAND vblank)
add_test(NAME cooperative_clock_vblank_late25 COMMAND vblank --late-opposite)
add_test(NAME cooperative_clock_vblank_late35 COMMAND vblank --late-opposite-35)
link_fixture(handoff "${CMAKE_CURRENT_SOURCE_DIR}/diagnostic/handoff_tests.cpp")
link_fixture(handoff_control "${CMAKE_CURRENT_SOURCE_DIR}/diagnostic/handoff_tests.cpp")
target_sources(handoff_control PRIVATE "${SOURCE}/ps2xRuntime/src/lib/Kernel/EeScheduler.cpp")
target_compile_definitions(handoff_control PRIVATE PS2X_VBLANK_HANDOFF=0 PS2X_VBLANK_HOST_ELIGIBILITY=0 EXPECT_MISSED_FIELDS=1)
foreach(mode IN ITEMS debt alarm multiple suspended suspend-ready delete-ready self-delete nested-wait nested-completion nested-missing nested-cancel polling priority stop no-waiters work35)
    add_test(NAME "cooperative_clock_handoff_${mode}" COMMAND handoff ${mode})
endforeach()
add_test(NAME cooperative_clock_handoff_disabled_control COMMAND handoff_control debt)

set(PROOF_INCLUDE "${CMAKE_CURRENT_BINARY_DIR}/proof-include")
file(READ "${SOURCE}/ps2xRuntime/include/runtime/ee_scheduler.h" SCHEDULER_HEADER)
set(ANCHOR "private:\n    struct ScheduledEvent")
string(FIND "${SCHEDULER_HEADER}" "${ANCHOR}" FOUND)
if(FOUND LESS 0)
    message(FATAL_ERROR "Review proof-only current scheduler friendship anchor")
endif()
string(REPLACE "${ANCHOR}" "private:\n    friend struct FieldModeProof;\n    friend struct PhysicalFieldProof;\n    struct ScheduledEvent" SCHEDULER_PROOF "${SCHEDULER_HEADER}")
file(MAKE_DIRECTORY "${PROOF_INCLUDE}/runtime")
file(WRITE "${PROOF_INCLUDE}/runtime/ee_scheduler.h" "${SCHEDULER_PROOF}")
configure_file("${SOURCE}/ps2xRuntime/include/runtime/vblank_field_pacer.h" "${PROOF_INCLUDE}/runtime/vblank_field_pacer.h" COPYONLY)
link_fixture(mode_api "${COMPAT}/vblank-tests/mode_tests.cpp")
target_include_directories(mode_api BEFORE PRIVATE "${PROOF_INCLUDE}")
target_include_directories(mode_api PRIVATE "${SOURCE}/ps2xRuntime/src/lib/Kernel")
add_test(NAME cooperative_clock_mode_api COMMAND mode_api)
link_fixture(physical_native "${CMAKE_CURRENT_SOURCE_DIR}/diagnostic/native_tests.cpp")
target_include_directories(physical_native BEFORE PRIVATE "${PROOF_INCLUDE}")
add_test(NAME cooperative_clock_physical_native COMMAND physical_native)

foreach(mode IN ITEMS no-wall-floor bulk-credit)
    execute_process(COMMAND "${Python3_EXECUTABLE}" "${COMPAT}/ee-clock-tests/prepare_clock_control.py"
        --source "${SOURCE}/ps2xRuntime/src/lib/Kernel/EeScheduler.cpp"
        --output "${CMAKE_CURRENT_BINARY_DIR}/EeScheduler-${mode}.cpp" --mode "${mode}"
        RESULT_VARIABLE result OUTPUT_VARIABLE out ERROR_VARIABLE err)
    if(NOT result EQUAL 0)
        message(FATAL_ERROR "Review current-layout ${mode} control: ${out}${err}")
    endif()
endforeach()
link_fixture(ee_clock "${COMPAT}/ee-clock-tests/ee_clock_tests.cpp")
add_test(NAME cooperative_clock_busy_native COMMAND ee_clock)
link_fixture(ee_clock_control "${COMPAT}/ee-clock-tests/ee_clock_tests.cpp")
target_sources(ee_clock_control PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/EeScheduler-no-wall-floor.cpp")
target_compile_definitions(ee_clock_control PRIVATE EXPECT_BOTH_CLOCKS_CONTROL=1)
add_test(NAME cooperative_clock_no_floor_control COMMAND ee_clock_control)
link_fixture(wall_clock "${CMAKE_CURRENT_SOURCE_DIR}/diagnostic/wall_clock_tests.cpp")
link_fixture(wall_clock_control "${CMAKE_CURRENT_SOURCE_DIR}/diagnostic/wall_clock_tests.cpp")
target_sources(wall_clock_control PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/EeScheduler-no-wall-floor.cpp")
target_compile_definitions(wall_clock_control PRIVATE EXPECT_CLOCK_UNDERCREDIT=1)
foreach(mode IN ITEMS normal fragmented long-isr masked no-handler reset-cycle idle instruction-ledger native-mmio)
    add_test(NAME "cooperative_clock_wall_${mode}" COMMAND wall_clock ${mode})
endforeach()
add_test(NAME cooperative_clock_native_undercredit_control COMMAND wall_clock_control normal)
link_fixture(chronological "${COMPAT}/ee-clock-tests/chronological_clock_tests.cpp")
link_fixture(bulk_control "${COMPAT}/ee-clock-tests/chronological_clock_tests.cpp")
target_sources(bulk_control PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/EeScheduler-bulk-credit.cpp")
target_compile_definitions(bulk_control PRIVATE EXPECT_LOST_ROLLOVERS=1)
foreach(mode IN ITEMS simple fragmented masked no-handler simultaneous multiple blocking alarm callback-wait callback-poll)
    add_test(NAME "cooperative_clock_chronological_${mode}" COMMAND chronological ${mode})
endforeach()
add_test(NAME cooperative_clock_bulk_credit_control COMMAND bulk_control fragmented)
get_property(all_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${all_tests} PROPERTIES TIMEOUT 30)

link_fixture(cooperative_scheduler "${CMAKE_CURRENT_SOURCE_DIR}/scheduler_admission_tests.cpp")
enable_testing()
foreach(mode IN ITEMS normal cancel fault idle poll lifecycle ahead reentry)
    add_test(NAME "cooperative_scheduler_${mode}" COMMAND cooperative_scheduler "${mode}")
endforeach()
set_tests_properties(cooperative_scheduler_normal cooperative_scheduler_cancel cooperative_scheduler_fault
    cooperative_scheduler_idle cooperative_scheduler_poll cooperative_scheduler_lifecycle cooperative_scheduler_ahead
    cooperative_scheduler_reentry PROPERTIES TIMEOUT 10)

link_fixture(w_admission "${CMAKE_CURRENT_SOURCE_DIR}/admission_tests.cpp")

foreach(mode normal cancel reset blocked)
    add_test(NAME "w_runtime_admission_${mode}" COMMAND w_admission ${mode})
endforeach()
get_property(all_tests DIRECTORY PROPERTY TESTS)
set_tests_properties(${all_tests} PROPERTIES TIMEOUT 30
    ENVIRONMENT "PS2X_GS_BACKEND=${COOPERATIVE_GS_BACKEND};PS2X_GS_THREADS=1;PS2X_GS_TRACE=;PS2X_GS_RECORD=")

    set_tests_properties(cooperative_clock_no_floor_control PROPERTIES DISABLED TRUE LABELS "obsolete-q-contract")
